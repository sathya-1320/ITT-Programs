class MyHashSet {
    private boolean[] exists;

    public MyHashSet() {
        exists = new boolean[1000001];
    }
    
    public void add(int key) {
        exists[key] = true;
    }
    
    public void remove(int key) {
        exists[key] = false;
    }
    
    public boolean contains(int key) {
        return exists[key];
    }
}
